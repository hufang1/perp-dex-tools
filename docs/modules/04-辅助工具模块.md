# 辅助工具模块

## 模块概述

辅助工具模块提供日志记录、消息通知等支持功能,为核心交易逻辑提供必要的基础设施。

## 核心文件

### 1. `helpers/logger.py`

日志记录系统,负责交易日志和调试日志的记录。

#### 1.1 `TradingLogger` 类

##### 初始化

```python
class TradingLogger:
    def __init__(
        self,
        exchange: str,
        ticker: str,
        log_to_console: bool = True
    ):
        """
        初始化交易日志记录器
        
        Args:
            exchange: 交易所名称
            ticker: 交易对符号
            log_to_console: 是否输出到控制台
        """
        self.exchange = exchange
        self.ticker = ticker
        self.account_name = os.getenv("ACCOUNT_NAME", "")
        
        # 创建日志目录
        self.log_dir = Path("logs") / exchange / ticker
        self.log_dir.mkdir(parents=True, exist_ok=True)
        
        # 设置日志文件路径
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        self.debug_log = self.log_dir / f"debug_{timestamp}.log"
        self.trade_log = self.log_dir / f"trades_{timestamp}.csv"
        
        # 初始化 CSV 文件
        self._init_trade_log()
        
        # 配置控制台输出
        self.log_to_console = log_to_console
```

##### 主要方法

**`log(message: str, level: str = "INFO")`**

记录通用日志信息:

```python
def log(self, message: str, level: str = "INFO"):
    """
    记录日志
    
    Args:
        message: 日志消息
        level: 日志级别 (INFO, WARNING, ERROR, DEBUG)
    """
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    # 构建日志消息
    if self.account_name:
        log_message = f"[{timestamp}] [{self.account_name}] [{level}] {message}"
    else:
        log_message = f"[{timestamp}] [{level}] {message}"
    
    # 写入文件
    with open(self.debug_log, 'a', encoding='utf-8') as f:
        f.write(log_message + '\n')
    
    # 控制台输出
    if self.log_to_console:
        if level == "ERROR":
            print(f"\033[91m{log_message}\033[0m")  # 红色
        elif level == "WARNING":
            print(f"\033[93m{log_message}\033[0m")  # 黄色
        else:
            print(log_message)
```

**特点**:
- **彩色输出**: 错误红色,警告黄色
- **时间戳**: 精确到秒
- **账户标识**: 支持多账户区分
- **文件持久化**: 所有日志写入文件

**`log_transaction(order_id, side, size, price, status)`**

记录交易信息到 CSV 文件:

```python
def log_transaction(
    self,
    order_id: str,
    side: str,
    size: Union[str, Decimal],
    price: Union[str, Decimal],
    status: str
):
    """
    记录交易到 CSV
    
    Args:
        order_id: 订单 ID
        side: 交易方向 (buy/sell)
        size: 交易数量
        price: 交易价格
        status: 订单状态
    """
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    
    with open(self.trade_log, 'a', encoding='utf-8', newline='') as f:
        writer = csv.writer(f)
        writer.writerow([
            timestamp,
            self.exchange,
            self.ticker,
            order_id,
            side,
            str(size),
            str(price),
            status
        ])
```

**CSV 格式**:
```csv

2023-12-15 14:30:45,edgex,ETH,12346,sell,0.1,3001.1,FILLED
```

**用途**:
- 交易历史分析
- 盈亏计算
- 策略回测
- 审计追踪

```python
def _init_trade_log(self):
    """创建 CSV 文件并写入表头"""
    if not self.trade_log.exists():
        with open(self.trade_log, 'w', encoding='utf-8', newline='') as f:
            writer = csv.writer(f)
            writer.writerow([
                'Timestamp',
                'Exchange',
                'Ticker',
                'Order ID',
                'Side',
                'Size',
                'Price',
                'Status'
            ])
```

```python
async def send_text(self, message: str):
    """
    发送文本消息
    
    Args:
        message: 消息内容
    """
    payload = {
        "msg_type": "text",
        "content": {
            "text": message
        }
    }
    
    async with aiohttp.ClientSession() as session:
        try:
            async with session.post(
                self.webhook_url,
                json=payload,
                timeout=aiohttp.ClientTimeout(total=10)

                return result.get('code') == 0
```

## 集成使用

### 在交易机器人中集成

```python
class TradingBot:
    def __init__(self, config):
        # 初始化日志
        self.logger = TradingLogger(
            exchange=config.exchange,
            ticker=config.ticker,
            log_to_console=True
        )
        
        # 初始化通知
        self.telegram_token = os.getenv("TELEGRAM_BOT_TOKEN")
        self.telegram_chat_id = os.getenv("TELEGRAM_CHAT_ID")
        self.lark_token = os.getenv("LARK_TOKEN")
    
    async def send_notification(self, message: str):
        """发送通知到所有配置的渠道"""
        # Lark 通知
        if self.lark_token:
            async with LarkBot(self.lark_token) as bot:
                await bot.send_text(message)
        
        # Telegram 通知
        if self.telegram_token and self.telegram_chat_id:
            with TelegramBot(self.telegram_token, self.telegram_chat_id) as bot:
                bot.send_text(message)
    
    async def handle_position_mismatch(self):
        """处理仓位不匹配"""
        error_message = f"""
        ERROR: [{self.config.exchange.upper()}_{self.config.ticker.upper()}]
        Position mismatch detected
        
        请手动平衡当前仓位和正在关闭的仓位
        Current position: {position}
        Active closing amount: {close_amount}
        
        机器人已停止交易
        """
        
        # 记录日志
        self.logger.log(error_message, "ERROR")
        
        # 发送通知
        await self.send_notification(error_message)
        
        # 停止交易
        self.shutdown_requested = True
```

## 日志分析工具

### 解析交易日志

```python
import pandas as pd

def analyze_trades(csv_file):
    """分析交易 CSV 文件"""
    df = pd.read_csv(csv_file)
    
    # 计算统计数据
    stats = {
        'total_trades': len(df),
        'buy_count': len(df[df['Side'] == 'buy']),
        'sell_count': len(df[df['Side'] == 'sell']),
        'total_volume': df['Size'].astype(float).sum(),
        'avg_price': df['Price'].astype(float).mean(),
        'filled_trades': len(df[df['Status'] == 'FILLED']),
        'canceled_trades': len(df[df['Status'] == 'CANCELED'])
    }
    
    return stats

# 使用示例
stats = analyze_trades('logs/edgex/ETH/trades_20231215_143022.csv')
print(f"总交易次数: {stats['total_trades']}")
print(f"总交易量: {stats['total_volume']}")
```

### 查找错误

```python
def find_errors(log_file):
    """查找日志中的错误"""
    errors = []
    
    with open(log_file, 'r') as f:
        for line in f:
            if '[ERROR]' in line:
                errors.append(line.strip())
    
    return errors

# 使用示例
errors = find_errors('logs/edgex/ETH/debug_20231215_143022.log')
for error in errors:
    print(error)
```

### 2. 消息去重

```python
class NotificationManager:
    def __init__(self):
        self.last_messages = {}
    
    async def send_once(self, key: str, message: str, timeout: int = 300):
        """在指定时间内只发送一次相同消息"""
        now = time.time()
        
        if key in self.last_messages:
            if now - self.last_messages[key] < timeout:
                return  # 跳过重复消息
        
        # 发送消息
        await self.send_notification(message)
        self.last_messages[key] = now
```

### 3. 性能考虑
